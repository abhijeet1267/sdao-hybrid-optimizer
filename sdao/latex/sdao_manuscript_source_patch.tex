\subsection*{Section 4.2: replacement mathematical formulation}
\label{sec:problem-formulation}

\begin{verbatim}
\subsection{Problem Formulation}
\label{sec:problem-formulation}

Let a hybrid query be
\[
q = \langle \mathbf{v}, P, K \rangle,
\]
where \(\mathbf{v}\) is the query embedding, \(P\) is a structured
predicate, and \(K\) is the requested result count. Let
\[
\sigma(P)
  = \frac{\left|\{x \in \mathcal{D} : P(x)\}\right|}
          {|\mathcal{D}|}
\]
denote the true predicate selectivity. For \(K' \gg K\), let
\[
\rho(\mathbf{v}, P)
  = \frac{
      \left|
        \mathcal{N}_{K'}(\mathbf{v})
        \cap \{x \in \mathcal{D} : P(x)\}
      \right|
    }{K'}
\]
denote the local overlap between predicate-satisfying tuples and the
\(K'\) nearest neighbors of \(\mathbf{v}\). Let
\[
\mathcal{S} = \{\PRE, \POST, \GRAPH, \ISECT\}.
\]
Given \(\widehat{\sigma}(P)\), \(\widehat{\rho}(\mathbf{v},P)\),
and \(\widehat{R}_{s}(q)\) for each \(s \in \mathcal{S}\), SDAO selects
\[
s^{\star}(q)
=
\underset{s \in \mathcal{S}}{\arg\min}\;
\mathbb{E}\!\left[
  \widehat{\operatorname{Cost}}
  \bigl(s \mid \widehat{\sigma}(P),
  \widehat{\rho}(\mathbf{v},P)\bigr)
\right]
\quad \text{subject to} \quad
\widehat{R}_{s}(q) \geq R_{\mathrm{target}}.
\]
\end{verbatim}

\subsection*{Section 4.9: replacement cost model}
\label{sec:cost-model}

\begin{verbatim}
\subsection{Cost Model}
\label{sec:cost-model}

Let \(\widehat{C}(P)\) be the estimated pre-filter candidate set:
\[
\left|\widehat{C}(P)\right| = \widehat{\sigma}(P)N.
\]
SDAO estimates the cost of each strategy as
\begin{align}
\widehat{\operatorname{Cost}}_{\PRE}
  &= c_1\left|\widehat{C}(P)\right|
   + c_2\left|\widehat{C}(P)\right|\log K, \\
\widehat{\operatorname{Cost}}_{\POST}
  &= c_3\operatorname{ANN}(Kf) + c_4Kf, \\
\widehat{\operatorname{Cost}}_{\GRAPH}
  &= c_5\,ef_{\mathrm{search}}
     \left(1 + \frac{1}{\max(\widehat{\sigma}(P),\varepsilon)}\right), \\
\widehat{\operatorname{Cost}}_{\ISECT}
  &= c_6\left(\left|B_v\right| + \left|B_p\right|\right),
\end{align}
where \(\varepsilon > 0\) prevents division by zero, \(f\) is the
post-filter oversampling factor, \(ef_{\mathrm{search}}\) is the graph
exploration factor, \(B_v\) is the selected vector bucket, and \(B_p\)
is the predicate bitmap. When \(\widehat{\rho}(\mathbf{v},P)\) is close
to \(\widehat{\sigma}(P)\), SDAO uses
\[
f \approx \frac{1}{\max(\widehat{\sigma}(P),\varepsilon)}.
\]
When \(\widehat{\rho}(\mathbf{v},P) < \widehat{\sigma}(P)\), the
planner increases \(f\) to compensate for the under-representation of
predicate-satisfying tuples near the query embedding.
\end{verbatim}
